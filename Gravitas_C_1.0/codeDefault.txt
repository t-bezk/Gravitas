#include <SFML/Graphics.hpp>
#include <SFML/Audio.hpp>
#include <SFML/Window.hpp>
#include <SFML/OpenGL.hpp>
#include <iostream>
#include <ctime>
#include <string>
#include <cstdlib>
#include <fstream>
int main()
{
	sf::Vector2i screen(800, 600);
	sf::RenderWindow Window(sf::VideoMode(screen.x, screen.y), "Window", sf::Style::Default);
	sf::Event event;
	sf::Clock clock;
	while (Window.isOpen())
	{
		while (Window.pollEvent(event))
		{
			switch (event.type)
			{
			case sf::Event::Closed:
			{
				Window.close();
				break;
			}
			
			}
		}
		Window.display();
		Window.clear();
	}
	return 0;
}
